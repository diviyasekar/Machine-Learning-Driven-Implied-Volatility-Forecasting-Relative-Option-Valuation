\section{Appendix A: Complete 39-Dimensional Feature Schema Specification}
\label{sec:appendix_feature_schema}

The production machine learning pipeline serializes an exact 39-dimensional feature schema in the model artifact file \path{nifty_iv_model_features.joblib}. Incoming option chain snapshots are aligned against this schema, stripping extraneous fields and applying median imputation to ensure fault-tolerant inference. Table \ref{tab:complete_feature_schema} outlines all 39 features, their temporal lag horizons, and their quantitative formulations.

import pandas as pd

def compute_exact_tte_and_lags(df: pd.DataFrame) -> pd.DataFrame:
    df = df.copy()
    # Standardized to 15:30 IST on weekly expiry
    df['expiry_dt'] = pd.to_datetime(df['expiry']) + pd.Timedelta(hours=15, minutes=30)
    df['exact_tte_hours'] = (df['expiry_dt'] - pd.to_datetime(df['timestamp'])).dt.total_seconds() / 3600.0
    df['exact_tte_hours'] = df['exact_tte_hours'].clip(lower=1e-5)
    
    # Sort and construct lags by contract key
    df = df.sort_values(by=['strike', 'option_type', 'timestamp'])
    grp = df.groupby(['strike', 'option_type'])
    df['IV_lag_5m'] = grp['IV'].shift(1)
    df['IV_lag_10m'] = grp['IV'].shift(2)
    df['IV_change_prev_5m'] = df['IV'] - df['IV_lag_5m']
    df['IV_change_prev_10m'] = df['IV'] - df['IV_lag_10m']
    df['bid_ask_spread_lag_5m'] = grp['bid_ask_spread'].shift(1)
    return df
\end{lstlisting}

\textbf{Listing B.2: Production Live Inference \& Valuation}
\begin{lstlisting}[language=Python]
import numpy as np

def run_live_inference_and_valuation(features, model, schema, medians, delta_val=0.50):
    # Align features against training schema
    X = features.reindex(columns=schema).fillna(medians)
    
    # Predict 5m volatility change and future IV
    pred_delta_iv = model.predict(X)
    current_iv = features['IV'].values
    pred_future_iv = current_iv + pred_delta_iv
    valuation_spread = current_iv - pred_future_iv
    
    # Assign institutional economic regime tags
    regimes = np.where(valuation_spread > delta_val, 'Rich / Overvalued',
              np.where(valuation_spread < -delta_val, 'Cheap / Undervalued', 'Fair / Neutral'))
    return pred_future_iv, pred_delta_iv, valuation_spread, regimes
\end{lstlisting}
