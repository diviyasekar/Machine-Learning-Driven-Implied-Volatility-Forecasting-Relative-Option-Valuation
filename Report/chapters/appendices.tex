\section{Appendix A: Mathematical Derivations of Greeks and Lemma 1}
\label{sec:appendix_lemma_proof}

This appendix provides the full calculus proofs for the Black-Scholes-Merton foundational identity (Lemma 1), the primary Greeks ($\Delta, \nu, \Theta$), and higher-order cross-sensitivities (Vanna, Volga, Charm).

\subsection{Proof of Lemma 1: The Boundary Identity $S N'(d_1) = K e^{-r\tau} N'(d_2)$}
Recall $N'(x) = \frac{1}{\sqrt{2\pi}} e^{-x^2 / 2}$ and $d_2 = d_1 - \sigma \sqrt{\tau}$. Evaluating the ratio:
\begin{align}
    \frac{S N'(d_1)}{K e^{-r\tau} N'(d_2)} &= \frac{S}{K e^{-r\tau}} \exp\left( -\tfrac{1}{2} (d_1^2 - d_2^2) \right) \\
    &= \frac{S}{K e^{-r\tau}} \exp\left( -\tfrac{1}{2} [2 d_1 \sigma \sqrt{\tau} - \sigma^2 \tau] \right) = \frac{S}{K e^{-r\tau}} \exp(-\ln(S/K) - r\tau) = 1
\end{align}
Hence, $S N'(d_1) = K e^{-r\tau} N'(d_2)$ holds identically for all $S, K, \sigma > 0$ and $\tau > 0$ \quad $\blacksquare$.

\subsection{Proof of Delta ($\Delta_C = N(d_1)$) and Vega ($\nu = S\sqrt{\tau}N'(d_1)$)}
Differentiating $C = S N(d_1) - K e^{-r\tau} N(d_2)$ with respect to $S$:
\begin{equation}
    \frac{\partial C}{\partial S} = N(d_1) + \left[ S N'(d_1) - K e^{-r\tau} N'(d_2) \right] \frac{\partial d_1}{\partial S} = N(d_1) \quad (\text{by Lemma 1})
\end{equation}
Differentiating $C$ with respect to $\sigma$ where $\frac{\partial d_2}{\partial \sigma} = \frac{\partial d_1}{\partial \sigma} - \sqrt{\tau}$:
\begin{equation}
    \frac{\partial C}{\partial \sigma} = \left[ S N'(d_1) - K e^{-r\tau} N'(d_2) \right] \frac{\partial d_1}{\partial \sigma} + K e^{-r\tau} N'(d_2) \sqrt{\tau} = S \sqrt{\tau} N'(d_1) \quad (\text{by Lemma 1})
\end{equation}

\subsection{Proof of Theta ($\Theta_C = -\frac{S \sigma N'(d_1)}{2\sqrt{\tau}} - r K e^{-r\tau} N(d_2)$)}
Differentiating $C$ with respect to $\tau$ where $\frac{\partial d_2}{\partial \tau} = \frac{\partial d_1}{\partial \tau} - \frac{\sigma}{2\sqrt{\tau}}$:
\begin{equation}
    \frac{\partial C}{\partial \tau} = \left[ S N'(d_1) - K e^{-r\tau} N'(d_2) \right] \frac{\partial d_1}{\partial \tau} + \frac{K e^{-r\tau} N'(d_2) \sigma}{2\sqrt{\tau}} + r K e^{-r\tau} N(d_2) = \frac{S \sigma N'(d_1)}{2\sqrt{\tau}} + r K e^{-r\tau} N(d_2)
\end{equation}
Since $\Theta_C = -\partial C / \partial \tau$, this establishes Eq. \eqref{eq:theta_call}.

\subsection{Derivations of Higher-Order Greeks (Vanna, Volga, Charm)}
\label{sec:appendix_higher_order_greeks}
\begin{align}
    \text{Vanna} &= \frac{\partial \Delta_C}{\partial \sigma} = N'(d_1) \frac{\partial d_1}{\partial \sigma} = -\frac{N'(d_1) d_2}{\sigma} = -\frac{e^{-d_1^2 / 2} d_2}{\sigma \sqrt{2\pi}} \label{eq:vanna_formula} \\
    \text{Volga} &= \frac{\partial \nu}{\partial \sigma} = S \sqrt{\tau} N'(d_1) \frac{d_1 d_2}{\sigma} = \nu \frac{d_1 d_2}{\sigma} \label{eq:volga_formula} \\
    \text{Charm} &= -\frac{\partial \Delta_C}{\partial \tau} = -N'(d_1) \left[ \frac{r}{\sigma \sqrt{\tau}} - \frac{d_2}{2\tau} \right] \label{eq:charm_formula}
\end{align}

\subsection{C.1 Exact Fractional TTE Calculation \& Vectorized Lag Construction}
\begin{lstlisting}[language=Python, caption={Exact Fractional TTE and Multi-Horizon Lag Construction.}]
import pandas as pd

def compute_exact_tte_and_lags(df: pd.DataFrame) -> pd.DataFrame:
    df = df.copy()
    # Fractional hours to expiry (standardized to 15:30 IST)
    df['expiry_dt'] = pd.to_datetime(df['expiry']) + pd.Timedelta(hours=15, minutes=30)
    df['exact_tte_hours'] = (df['expiry_dt'] - pd.to_datetime(df['timestamp'])).dt.total_seconds() / 3600.0
    df['exact_tte_hours'] = df['exact_tte_hours'].clip(lower=1e-5)
    
    # Sort and construct autoregressive lags by contract key
    df = df.sort_values(by=['strike', 'option_type', 'timestamp'])
    grp = df.groupby(['strike', 'option_type'])
    df['IV_lag_5m'] = grp['IV'].shift(1)
    df['IV_lag_10m'] = grp['IV'].shift(2)
    df['IV_change_prev_5m'] = df['IV'] - df['IV_lag_5m']
    df['IV_change_prev_10m'] = df['IV'] - df['IV_lag_10m']
    df['bid_ask_spread_lag_5m'] = grp['bid_ask_spread'].shift(1)
    return df
\end{lstlisting}

\subsection{C.2 Production Live Inference \& Relative Valuation Screener}
\begin{lstlisting}[language=Python, caption={Production Live Inference and Valuation Categorization.}]
import numpy as np

def run_live_inference_and_valuation(current_features, model, schema, medians, delta_val=0.50):
    X = current_features.reindex(columns=schema).fillna(medians)
    pred_delta_iv = model.predict(X)
    current_iv = current_features['IV'].values
    pred_future_iv = current_iv + pred_delta_iv
    valuation_spread = current_iv - pred_future_iv
    
    # Economic regime tags
    regimes = np.where(valuation_spread > delta_val, 'Rich / Overvalued',
              np.where(valuation_spread < -delta_val, 'Cheap / Undervalued', 'Fair / Neutral'))
    return pred_future_iv, pred_delta_iv, valuation_spread, regimes
\end{lstlisting}
